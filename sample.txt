QTY_WORDS = {"a": 1, "an": 1, "one": 1, "two": 2, "to": 2, "too": 2, "three": 3,
             "four": 4, "five": 5, "six": 6, "seven": 7, "eight": 8, "nine": 9,
             "ten": 10, "eleven": 11, "twelve": 12}
DIGIT_WORDS = {"zero": "0", "oh": "0", "one": "1", "two": "2", "three": "3",
               "four": "4", "five": "5", "six": "6", "seven": "7", "eight": "8",
               "nine": "9"}
EXTRA_NAMES = {"choc": "Chocolate Cake", "velvet": "Red Velvet Cake",
               "cup cake": "Cupcake", "cup cakes": "Cupcake"}
PHONE_RE = r"(?:\+?254|0)[\s\-]?[71](?:[\s\-]?\d){8}"


def menu_aliases():
    aliases = {}
    for name in MENU:
        low = name.lower()
        forms = {low}
        if low.endswith(" cake"):
            forms.add(low[:-5])
        for f in list(forms):
            forms.add(f + "s")
        for f in forms:
            aliases[f] = name
    for word, name in EXTRA_NAMES.items():
        if name in MENU:
            aliases[word] = name
    return aliases


def find_phone(text):
    tokens = re.findall(r"[a-z]+|\d+", (text or "").lower())
    joined = " ".join(DIGIT_WORDS.get(t, t) for t in tokens)
    m = re.search(r"(?:254|0)\s?[71](?:\s?\d){8}", joined)
    return clean_phone(m.group().replace(" ", "")) if m else None


def parse_items(text):
    work = re.sub(r"[^a-z0-9\s]", " ", (text or "").lower())
    work = re.sub(PHONE_RE, " ", work)
    work = re.sub(r"\s+", " ", work)
    words = "|".join(QTY_WORDS)
    found = {}
    aliases = menu_aliases()
    for alias in sorted(aliases, key=len, reverse=True):
        pat = re.compile(r"\b(?:(\d+|" + words + r")\s+(?:of\s+)?)?" + re.escape(alias) + r"\b")
        while True:
            m = pat.search(work)
            if not m:
                break
            q = m.group(1)
            qty = int(q) if q and q.isdigit() else QTY_WORDS.get(q, 1)
            name = aliases[alias]
            found[name] = min(50, found.get(name, 0) + max(1, qty))
            work = work[:m.start()] + " " * (m.end() - m.start()) + work[m.end():]
    return found


def voice_command(text):
    low = (text or "").lower()
    if re.search(r"\b(clear|cancel|reset|start over|empty)\b", low):
        return "clear"
    if re.search(r"\b(pay|checkout|check out|place order|confirm)\b", low):
        return "pay"
    return None


@app.route("/voice", methods=["POST"])
def voice():
    text = (request.get_json(silent=True) or {}).get("text", "")
    return jsonify(items=parse_items(text), phone=find_phone(text),
                   command=voice_command(text), heard=text)
